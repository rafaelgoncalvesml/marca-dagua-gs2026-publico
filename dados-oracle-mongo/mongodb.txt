// Consulta 2: eventos em ALERTA do dispositivo MD-POA-001, do mais recente ao mais antigo
db.eventos_brutos.find(
  { device_id: "MD-POA-001", risco: "ALERTA" },
  { _id: 0, device_id: 1, seq: 1, ts: 1, nivel_cm: 1, chuva_mm_h: 1, risco: 1 }
).sort({ ts: -1, seq: -1 })

// Consulta 3: por dispositivo e risco, total de eventos, nível máximo e chuva média
db.eventos_brutos.aggregate([
  { $group: { _id: { device_id: "$device_id", risco: "$risco" },
              total: { $sum: 1 },
              nivel_max_cm: { $max: "$nivel_cm" },
              chuva_media_mm_h: { $avg: "$chuva_mm_h" } } },
  { $sort: { "_id.device_id": 1, "_id.risco": 1 } }
])

// Consulta 4 (cruzamento com o Oracle): relatos de moradores junto do pior
// evento do mesmo device_id. No Oracle, o mesmo código é dispositivo.codigo.
db.relatos.aggregate([
  { $lookup: { from: "eventos_brutos", localField: "device_id",
               foreignField: "device_id", as: "eventos" } },
  { $project: { _id: 0, device_id: 1, tipo: 1, local: 1,
                eventos_no_dispositivo: { $size: "$eventos" },
                nivel_max_cm: { $max: "$eventos.nivel_cm" } } }
])
